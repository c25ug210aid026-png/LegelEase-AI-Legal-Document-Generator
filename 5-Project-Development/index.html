Both parties agree to the terms mentioned.\n2. This document is legally binding.\n3. Generated by LegalEase AI System.\n\nSignatures:\n${p1} ___________\n${p2} ___________`; document.getElementById('output').innerText=text; } function downloadPDF(){ const {jsPDF}=window.jspdf; let doc=new jsPDF(); let text=document.getElementById('output').innerText; doc.text(text,10,20); doc.save("Legal_Document.pdf"); }
